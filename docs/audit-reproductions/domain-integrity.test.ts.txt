import { randomUUID } from 'node:crypto';
import { afterAll, beforeEach, expect, it } from 'vitest';
import { eq, sql } from 'drizzle-orm';
import { db, pool, unitTransaction } from '../../src/server/db';
import { assignments, balances, duties, ledger, records, soldiers, soldierContacts } from '../../src/server/schema';
import { createInvitedAccount, type Actor } from '../../src/server/auth/accounts';
import { executeAction } from '../../src/server/actions';
import { postScore, settleDue } from '../../src/server/scoring';
import { soldier } from '../fixtures';

if (!process.env.DATABASE_URL?.endsWith('/fair_shifts_domain_audit_test') || process.env.DATABASE_URL !== process.env.TEST_DATABASE_URL) throw new Error('Dedicated domain audit test database required');
const DAY = 86400000;
let manager: Actor;
let technical: Actor;
async function person(role: 'soldier' | 'manager', name: string): Promise<Actor> {
 const id = randomUUID();
 const email = `${id}@example.invalid`;
 await db.insert(soldiers).values({id, name, data: soldier({id, name})});
 await db.insert(soldierContacts).values({soldierId:id,email});
 await db.insert(balances).values({soldierId:id});
 const account = await createInvitedAccount({name,email,soldierId:id,role});
 return {id:account.id,name,role,soldierId:id,securityEpoch:account.securityEpoch};
}
async function command(type:string,payload:Record<string,unknown>,version?:number,actor=manager) {
 return await executeAction(actor,{type,payload,expectedVersion:version,idempotencyKey:randomUUID()}) as any;
}
async function currentBalance(id:string) {
 return (await db.select().from(balances).where(eq(balances.soldierId,id)))[0].current;
}
beforeEach(async()=> {
 await db.execute(sql`truncate table auth_user, auth_budget, auth_rate_limit, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`);
 manager=await person('manager','Audit manager');
 const account=await createInvitedAccount({name:'Audit technical',email:'audit-technical@example.invalid',role:'technical'});
 technical={id:account.id,name:account.name,role:'technical',securityEpoch:account.securityEpoch};
});
afterAll(async()=>pool.end());
it('requires a score decision when a date-only correction crosses normalization',async()=> {
 const member=await person('soldier','Audit member');
 const type=await command('dutyType.save',{name:'Audit fixed duty',pricing:{mode:'fixed',base:10},roles:[{name:'Seat',count:1}]});
 const created=await command('duty.create',{typeId:type.id,name:'Audit duty',start:new Date(Date.now()+DAY).toISOString(),end:new Date(Date.now()+2*DAY).toISOString()});
 let duty=(await db.select().from(duties).where(eq(duties.id,created.id)))[0];
 await command('duty.assign',{dutyId:duty.id,slotId:duty.data.slots[0].id,soldierId:member.soldierId},duty.version);
 duty=(await db.select().from(duties).where(eq(duties.id,created.id)))[0];
 await command('duty.publish',{id:duty.id,confirmed:true},duty.version);
 duty=(await db.select().from(duties).where(eq(duties.id,created.id)))[0];
 const start=new Date(Date.now()-3*DAY).toISOString();
 const end=new Date(Date.now()-2*DAY).toISOString();
 await db.update(duties).set({data:{...duty.data,start,end}}).where(eq(duties.id,duty.id));
 await unitTransaction(tx=>settleDue(tx));
 await unitTransaction(tx=>postScore(tx,{soldierId:member.soldierId!,sourceKey:'audit-normalization',kind:'normalization',actorId:manager.id,reason:'Audit normalization',absolute:5,effectiveAt:new Date(Date.now()-DAY),data:{barrier:true}}));
 const assignment=(await db.select().from(assignments).where(eq(assignments.dutyId,duty.id)))[0];
 const input={assignmentId:assignment.id,start,end:new Date(Date.now()-3600000).toISOString(),reason:'Correct end date with unchanged fixed price'};
 const preview=await command('performance.correction.preview',input,assignment.version);
 const saved=await command('performance.correction.apply',{...input,token:preview.token},assignment.version);
 const pending=(await db.select().from(records).where(eq(records.kind,'score_decision'))).filter(r=>r.data.status==='pending');
 console.log(JSON.stringify({scenario:'date-only crossing normalization',previewEffects:preview.effects,outcomes:saved.outcomes,pending:pending.length,balance:await currentBalance(member.soldierId!)}));
 expect(pending).toHaveLength(1);
});
it('keeps a frozen balance unchanged when promotion occurs after score preview',async()=> {
 const member=await person('soldier','Audit promoted member');
 await unitTransaction(tx=>postScore(tx,{soldierId:member.soldierId!,sourceKey:'audit-opening',kind:'opening',actorId:manager.id,reason:'Audit opening',absolute:10,effectiveAt:new Date()}));
 const input={soldierIds:[member.soldierId!],operation:'subtract',value:4,reason:'Audit score change'};
 const preview=await command('score.preview',input);
 await command('account.role',{id:member.id,role:'manager'},member.securityEpoch,technical);
 let errorCode:string|undefined;
 try { await command('score.apply',{...input,token:preview.token}); } catch (error) { errorCode=(error as {code?:string}).code; }
 const after=await currentBalance(member.soldierId!);
 console.log(JSON.stringify({scenario:'promotion after score preview',errorCode,before:10,after}));
 expect(after).toBe(10);
 expect(errorCode).toBe('manager_balance_frozen');
});
