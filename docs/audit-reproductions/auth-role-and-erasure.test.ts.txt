import { randomUUID } from "node:crypto";
import { afterAll, beforeEach, describe, expect, it } from "vitest";
import { eq, sql } from "drizzle-orm";
import { db, pool } from "../../src/server/db";
import { user } from "../../src/server/auth-schema";
import { balances, records, soldierContacts, soldiers } from "../../src/server/schema";
import { createInvitedAccount, type Actor } from "../../src/server/auth/accounts";
import { executeAction } from "../../src/server/actions";
import { soldier, testIdNumber, testProfile } from "../fixtures";

// This reproduction truncates its dedicated database; never use application data.
if (!process.env.TEST_DATABASE_URL || process.env.TEST_DATABASE_URL !== process.env.DATABASE_URL || new URL(process.env.TEST_DATABASE_URL).pathname !== "/fair_shifts_auth_audit_test")
  throw new Error("Dedicated auth audit test database required");

let technical: Actor;
let manager: Actor;
let otherManager: Actor;
let target: Actor;
let sequence = 1;

async function invite(role: "soldier" | "manager"): Promise<Actor> {
  const soldierId = randomUUID();
  const serial = sequence++;
  const name = `Audit Person ${serial}`;
  const email = `audit-person-${serial}@example.invalid`;
  await db.insert(soldiers).values({
    id: soldierId,
    name,
    idNumber: testIdNumber(serial),
    data: soldier({ id: soldierId, name, ...testProfile(name) }),
  });
  await db.insert(soldierContacts).values({ soldierId, email, phone: "0500000001", address: "Synthetic private address" });
  await db.insert(balances).values({ soldierId, current: 0 });
  const row = await createInvitedAccount({ name, role, email, soldierId });
  return { id: row.id, name, role, soldierId, securityEpoch: row.securityEpoch };
}

async function refreshed(actor: Actor): Promise<Actor> {
  const [row] = await db.select().from(user).where(eq(user.id, actor.id));
  return { id: row.id, name: row.name, role: row.role as Actor["role"], soldierId: row.soldierId ?? undefined, securityEpoch: row.securityEpoch };
}

async function send(actor: Actor, type: string, payload: Record<string, unknown>, expectedVersion?: number, idempotencyKey = randomUUID()) {
  return executeAction(actor, { type, payload, expectedVersion, idempotencyKey }) as Promise<Record<string, unknown>>;
}

beforeEach(async () => {
  await db.execute(sql`truncate table auth_user, auth_budget, auth_rate_limit, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`);
  sequence = 1;
  const tech = await createInvitedAccount({ name: "Audit Technical", role: "technical", email: "audit-technical@example.invalid" });
  technical = { id: tech.id, name: tech.name, role: "technical", securityEpoch: tech.securityEpoch };
  manager = await invite("manager");
  otherManager = await invite("manager");
  target = await invite("soldier");
});

afterAll(async () => pool.end());

describe("pre-production audit regressions", () => {
  it("does not replay manager-only import data after that account is demoted", async () => {
    const [person] = await db.select().from(soldiers).where(eq(soldiers.id, target.soldierId!));
    const batch = await send(manager, "import.preview", { filename: "audit.xlsx", rows: [{ rowNumber: 2, values: { idNumber: person.idNumber, phone: "0500000002" } }] });
    const payload = { id: batch.id };
    const key = randomUUID();
    const original = await send(manager, "import.get", payload, undefined, key);
    expect(JSON.stringify(original)).toContain("Synthetic private address");
    await send(technical, "account.role", { id: manager.id, role: "soldier" }, manager.securityEpoch);
    const demoted = await refreshed(manager);
    await expect(send(demoted, "import.get", payload)).rejects.toMatchObject({ status: 403 });
    // Same operation and key must also apply the current permission boundary.
    await expect(send(demoted, "import.get", payload, undefined, key)).rejects.toMatchObject({ status: 403 });
  });

  it("can delete a former manager who has an email-change request", async () => {
    await send(manager, "manager.email.request", { email: "audit-manager-next@example.invalid", reason: "Synthetic audit request" });
    expect((await db.select().from(records).where(eq(records.kind, "manager_email_change"))).length).toBe(1);
    await send(technical, "account.role", { id: manager.id, role: "soldier" }, manager.securityEpoch);
    const [person] = await db.select().from(soldiers).where(eq(soldiers.id, manager.soldierId!));
    const preview = await send(otherManager, "soldier.delete.preview", { id: person.id }, person.version);
    await expect(send(otherManager, "soldier.delete", { id: person.id, previewToken: preview.previewToken, confirmed: true, reason: "Synthetic audit deletion" }, person.version)).resolves.toMatchObject({ id: person.id });
  });
});
