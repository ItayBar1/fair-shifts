/*
 * Synthetic reproduction for the production audit of 2026-10-18.
 * No database, real credentials, network requests or persistent writes.
 * The injected Drive fixture exposes a signed log in a different folder.
 * Exit 0 means the current cross-folder lookup flaw was reproduced.
 * After the lookup is fixed, the assertions should fail until this evidence
 * script is converted into a regression that expects no foreign files.
 *
 * Run from the repository root with the existing audit tooling image:
 * docker run --rm -i --network none --read-only --tmpfs /tmp --cap-drop ALL \
 *   --security-opt no-new-privileges -v "$PWD/src:/app/src:ro" \
 *   fs-audit-20261018-tests node --import tsx --input-type=module \
 *   < docs/audit-reproductions/drive-scope.ts.txt
 */
import assert from "node:assert/strict";
import { createHash, generateKeyPairSync } from "node:crypto";
import { driveStorage } from "./src/server/operations/backup-storage.ts";
import {
  nextEntry,
  parseLog,
  serializeEntry,
} from "./src/domain/deletion-log.ts";

const otherKeys = generateKeyPairSync("ed25519");
const selectedKeys = generateKeyPairSync("ed25519");
const otherLog = serializeEntry(
  nextEntry(
    undefined,
    {
      id: "00000000-0000-4000-8000-000000000001",
      soldierId: "00000000-0000-4000-8000-000000000002",
      at: "2026-10-18T00:00:00.000Z",
    },
    { keyId: "other-key", privateKey: otherKeys.privateKey },
  ),
);
assert.deepEqual(parseLog(otherLog, { "other-key": otherKeys.publicKey }).problems, []);

const queries = [];
const storage = driveStorage(
  {
    clientId: "synthetic-client",
    clientSecret: "synthetic-secret",
    refreshToken: "synthetic-token",
    folderId: "selected-folder",
  },
  async (input) => {
    const url = new URL(String(input));
    if (url.hostname === "oauth2.googleapis.com" && url.pathname === "/token") {
      return Response.json({ access_token: "synthetic-access-token", expires_in: 3600 });
    }
    assert.equal(url.hostname, "www.googleapis.com");
    if (url.pathname.endsWith("/files/other-log") && url.searchParams.get("alt") === "media") {
      return new Response(otherLog);
    }
    if (url.pathname.endsWith("/files")) {
      const query = url.searchParams.get("q");
      assert.ok(query);
      queries.push(query);
      return Response.json({
        files: query.includes("'selected-folder' in parents")
          ? []
          : [{
              id: "other-log",
              size: String(Buffer.byteLength(otherLog)),
              sha256Checksum: createHash("sha256").update(otherLog).digest("hex"),
            }],
      });
    }
    throw new Error("Unexpected synthetic Drive request");
  },
);

const found = await storage.findKind("deletion-log");
assert.equal(found.length, 1);
assert.equal(found[0].id, "other-log");
assert.ok(!queries[0].includes("'selected-folder' in parents"));
const content = await storage.read(found[0].id);
assert.ok(content);
const problems = parseLog(content.toString(), { "selected-key": selectedKeys.publicKey }).problems;
assert.deepEqual(problems, [{ code: "signature", line: 1 }]);

// The ordinary backup listing already applies the selected folder boundary.
assert.deepEqual(await storage.listBackups(), []);
assert.ok(queries[1].includes("'selected-folder' in parents"));
console.log(JSON.stringify({
  kindQuery: queries[0],
  returnedFiles: found.map((file) => file.id),
  selectedKeyProblems: problems,
  backupQueryIsScoped: true,
}));
