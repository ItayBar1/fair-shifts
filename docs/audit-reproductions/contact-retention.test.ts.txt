import { afterAll, beforeEach, expect, it } from "vitest";
import { eq, sql } from "drizzle-orm";
import { db, pool } from "../../src/server/db";
import { contactRequests, emailOutbox } from "../../src/server/schema";
import { createInvitedAccount } from "../../src/server/auth/accounts";
import { refreshContactDigest, submitContactRequest } from "../../src/server/contact-requests";
import { deliverNextEmail } from "../../src/server/operations/email";

// This reproduction clears its database. Never run it against application data.
if (
  !process.env.TEST_DATABASE_URL ||
  process.env.TEST_DATABASE_URL !== process.env.DATABASE_URL ||
  new URL(process.env.TEST_DATABASE_URL).pathname !== "/fair_shifts_contact_audit_test"
)
  throw new Error("Dedicated contact audit test database required");

beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, auth_budget, auth_rate_limit, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results, contact_requests cascade`
  );
  await createInvitedAccount({
    name: "Synthetic Contact Operator",
    role: "technical",
    email: "contact-operator@example.invalid",
  });
});

afterAll(async () => pool.end());

it("removes public contact content from the stored mail after its retention period", async () => {
  // Next daytime stays within the mail queue's one-day delivery window.
  const wallClock = Date.now();
  const sentAt = new Date(wallClock);
  sentAt.setUTCHours(9, 0, 0, 0);
  if (sentAt.getTime() <= wallClock)
    sentAt.setUTCDate(sentAt.getUTCDate() + 1);
  const input = {
    name: "Synthetic Contact Person",
    contact: "synthetic-contact@example.invalid",
    message: "Synthetic private request for retention verification",
  };
  await submitContactRequest(
    new Request("http://localhost/api/contact", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify(input),
    }),
    sentAt
  );
  expect(await db.transaction((tx) => refreshContactDigest(tx, sentAt))).toBe(1);
  const delivered = await deliverNextEmail(async (message) => {
    expect(message.text).toContain(input.message);
    return "synthetic-provider-id";
  }, sentAt);
  expect(delivered).toMatchObject({ status: "sent" });

  await db.transaction((tx) =>
    refreshContactDigest(tx, new Date(sentAt.getTime() + 31 * 86_400_000))
  );
  expect(await db.select().from(contactRequests)).toHaveLength(0);
  const retained = await db
    .select()
    .from(emailOutbox)
    .where(eq(emailOutbox.kind, "contact-digest"));
  // A delivery tombstone may remain, but the promise covers all stored content.
  expect(JSON.stringify(retained)).not.toContain(input.message);
  expect(JSON.stringify(retained)).not.toContain(input.contact);
  expect(JSON.stringify(retained)).not.toContain(input.name);
});
