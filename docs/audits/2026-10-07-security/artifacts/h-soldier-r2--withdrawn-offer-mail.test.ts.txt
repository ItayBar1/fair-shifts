// Audit harness h-soldier-r2: do withdrawn transfer offers still send and charge mail quota?
const DAY = 86_400_000;
let mgr: Actor;
let alon: Actor;
let cands: Actor[];

async function invite(name: string, role: "soldier" | "manager", pn: string) {
  const soldierId = randomUUID();
  await db.insert(soldiers).values({
    id: soldierId,
    name,
    personalNumber: pn,
    data: soldier({ id: soldierId, name, personalNumber: pn }),
  });
  await db
    .insert(soldierContacts)
    .values({ soldierId, email: `${pn}@example.invalid` });
  await db.insert(balances).values({ soldierId });
  const row = await createInvitedAccount({
    name,
    role,
    email: `${pn}@example.invalid`,
    soldierId,
  });
  return { id: row.id, name, role, soldierId, securityEpoch: 1 } as Actor;
}
async function command(
  actor: Actor,
  type: string,
  payload: Record<string, unknown>,
  expectedVersion?: number
) {
  return (await executeAction(actor, {
    type,
    payload,
    expectedVersion,
    idempotencyKey: randomUUID(),
  })) as { id: string; version: number } & Record<string, unknown>;
}
async function dutyRow(id: string) {
  const [row] = await db.select().from(duties).where(eq(duties.id, id));
  return row;
}

beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`
  );
  await db.execute(sql`delete from email_outbox`);
  mgr = await invite("manager", "manager", "40002");
  alon = await invite("alon", "soldier", "40001");
  cands = [
    await invite("c1", "soldier", "40011"),
    await invite("c2", "soldier", "40012"),
    await invite("c3", "soldier", "40013"),
  ];
});
afterAll(async () => pool.end());

describe("h-soldier-r2 withdrawn offer mail", () => {
  it("offer/withdraw cycles leave offer mail that is still sent and charged", async () => {
    const type = await command(mgr, "dutyType.save", {
      name: "type x",
      pricing: { mode: "fixed", base: 4 },
      roles: [{ name: "role", count: 1 }],
    });
    const duty = await command(mgr, "duty.create", {
      typeId: type.id,
      name: "duty x",
      start: new Date(Date.now() + 3 * DAY).toISOString(),
      end: new Date(Date.now() + 4 * DAY).toISOString(),
    });
    const created = await dutyRow(duty.id);
    await command(mgr, "duty.assign", { dutyId: duty.id, slotId: created.data.slots[0].id, soldierId: alon.soldierId, callUpBonus: 0 }, 1);
    await command(mgr, "duty.publish", { id: duty.id, confirmed: true }, 2);
    const [seat] = await db.select().from(assignments).where(eq(assignments.dutyId, duty.id));
    // Mail already spent today by other unit traffic, leaving 2 non-code sends.
    const day = quotaDay(new Date());
    await db.insert(emailQuota).values({ day, used: DAILY_QUOTA - CODE_RESERVE - 2 });
    const cycles = 2;
    for (let i = 0; i < cycles; i++) {
      const offer = await command(alon, "transfer.offer", { assignmentId: seat.id, candidateIds: cands.map((c) => c.soldierId) }, seat.version);
      const [row] = await db.select().from(records).where(eq(records.id, offer.id));
      await command(alon, "transfer.withdraw", { id: offer.id }, row.version);
    }
    // A later publication mail for one of the candidates (unit traffic).
    await db.transaction((tx) =>
      enqueueEmail(tx, {
        recipientAccountId: cands[0].id,
        eventKey: `synthetic-publication:${randomUUID()}`,
        kind: "publication",
        title: "synthetic publication",
        body: "synthetic",
        priority: 1,
        expiresAt: new Date(Date.now() + 3600_000),
      })
    );
    const queuedBefore = await db.select().from(emailOutbox);
    const sent: string[] = [];
    const transport = async (m: { eventKey: string }) => {
      sent.push(m.eventKey);
      return `fake-${sent.length}`;
    };
    const statuses: string[] = [];
    for (let i = 0; i < 12; i++) {
      const r = await deliverNextEmail(transport);
      statuses.push(r.status);
      if (r.status === "idle") break;
    }
    const outbox = await db.select().from(emailOutbox);
    const [quota] = await db.select().from(emailQuota).where(eq(emailQuota.day, day));
    const requests = (await db.select().from(records).where(eq(records.kind, "request"))).map((r) => r.data.status);
    const publication = outbox.find((m) => m.kind === "publication");
    const result = {
      requestStatuses: requests,
      transferMailQueued: queuedBefore.filter((m) => m.kind === "transfer").length,
      sentEventKeys: sent,
      deliveryStatuses: statuses,
      quotaUsed: quota.used,
      nonCodeLimit: DAILY_QUOTA - CODE_RESERVE,
      publicationStatus: publication?.status,
      publicationError: publication?.error,
      transferStillPending: outbox.filter((m) => m.kind === "transfer" && m.status === "pending").length,
    };
    console.log("WITHDRAWN_OFFER_MAIL " + JSON.stringify(result));
    expect(requests.every((s) => s === "cancelled")).toBe(true);
    expect(sent.every((key) => key.startsWith("transfer:"))).toBe(true);
    expect(sent.length).toBe(2);
    expect(publication?.status).toBe("pending");
  });
});
