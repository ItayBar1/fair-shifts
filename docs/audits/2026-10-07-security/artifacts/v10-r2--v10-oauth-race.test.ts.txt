// Independent verifier (v10-r2) reproduction. Synthetic data only, Google's
// token endpoint stubbed, no outbound network. A verified email change
// (applyVerifiedEmailChange = Google disconnect) is started from a separate
// async task and commits on its own DB connection while a Google callback is
// paused right after one of Better Auth's own adapter reads/writes.
import { randomUUID } from "node:crypto";
import { afterAll, beforeEach, describe, expect, it, vi } from "vitest";
import { and, eq, sql } from "drizzle-orm";
import { db, pool } from "../src/server/db";
import { user, account, session } from "../src/server/auth-schema";
import { soldiers, soldierContacts, balances } from "../src/server/schema";
import {
  applyVerifiedEmailChange,
  createInvitedAccount,
} from "../src/server/auth/accounts";
import { getActor, getAuth } from "../src/server/auth";
import { soldier } from "../tests/fixtures";

const clientId = "synthetic-client.apps.googleusercontent.com";
process.env.GOOGLE_CLIENT_ID = clientId;
process.env.GOOGLE_CLIENT_SECRET = "synthetic-google-client-secret";
const base = process.env.BETTER_AUTH_URL ?? "http://localhost:3000";

type GoogleProfile = { sub: string; email: string; email_verified: boolean };
const profiles = new Map<string, GoogleProfile>();
const seg = (v: object) => Buffer.from(JSON.stringify(v)).toString("base64url");
function idToken(p: GoogleProfile) {
  const now = Math.floor(Date.now() / 1000);
  return [
    seg({ alg: "RS256", typ: "JWT", kid: "synthetic" }),
    seg({ iss: "https://accounts.google.com", aud: clientId, iat: now, exp: now + 3600, name: "v10", ...p }),
    "synthetic-signature",
  ].join(".");
}
const realFetch = globalThis.fetch;
vi.stubGlobal("fetch", async (input: string | URL | Request, init?: RequestInit) => {
  const url = typeof input === "string" ? input : input instanceof URL ? input.href : input.url;
  if (url.startsWith("https://oauth2.googleapis.com/token")) {
    const p = profiles.get(new URLSearchParams(String(init?.body ?? "")).get("code") ?? "");
    if (!p) return Response.json({ error: "invalid_grant" }, { status: 400 });
    return Response.json({ access_token: "synthetic-access", token_type: "Bearer", expires_in: 3600, scope: "openid email profile", id_token: idToken(p) });
  }
  if (url.startsWith("https://")) throw new Error(`Unexpected request: ${url}`);
  return realFetch(input, init);
});
const cookiesOf = (r: Response) => r.headers.getSetCookie().map((c) => c.split(";")[0]).join("; ");
async function googleSignIn(p: GoogleProfile) {
  const start = await getAuth().handler(new Request(`${base}/api/auth/sign-in/social`, {
    method: "POST",
    headers: { origin: base, "content-type": "application/json" },
    body: JSON.stringify({ provider: "google", callbackURL: "/", errorCallbackURL: "/login" }),
  }));
  const { url } = (await start.json()) as { url: string };
  const state = new URL(url).searchParams.get("state")!;
  const code = randomUUID();
  profiles.set(code, p);
  const cb = await getAuth().handler(new Request(
    `${base}/api/auth/callback/google?code=${code}&state=${encodeURIComponent(state)}`,
    { headers: { cookie: cookiesOf(start) } }));
  const loc = new URL(cb.headers.get("location") ?? "", base);
  const cookie = cookiesOf(cb);
  return { signedIn: cb.status === 302 && loc.pathname === "/", error: loc.searchParams.get("error"), cookie: cookie.includes("session_token") ? cookie : "" };
}

const OLD = "v10-old@example.invalid";
const NEXT = "v10-next@example.invalid";
beforeEach(async () => {
  await db.execute(sql`truncate table auth_user, auth_verification, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`);
  profiles.clear();
  vi.restoreAllMocks();
});
afterAll(async () => { vi.unstubAllGlobals(); await pool.end(); });

async function inviteSoldier() {
  const id = randomUUID();
  await db.insert(soldiers).values({ id, name: "v10", personalNumber: "4200001", data: soldier({ id, name: "v10", personalNumber: "4200001" }) });
  await db.insert(soldierContacts).values({ soldierId: id, email: OLD });
  await db.insert(balances).values({ soldierId: id });
  return (await createInvitedAccount({ name: "v10", email: OLD, soldierId: id })).id;
}
const links = (uid: string) => db.select({ accountId: account.accountId }).from(account).where(and(eq(account.userId, uid), eq(account.providerId, "google")));
const row = async (uid: string) => (await db.select().from(user).where(eq(user.id, uid)))[0];

/**
 * Pauses the callback right after Better Auth's own adapter call returns,
 * starts the verified email change as an independent task, and resumes the
 * callback only once that change has committed.
 */
async function pauseAfter(method: "updateAccount" | "findUserByEmail", uid: string) {
  const ctx = await getAuth().$context;
  const adapter = ctx.internalAdapter as unknown as Record<string, (...a: unknown[]) => Promise<unknown>>;
  const original = adapter[method].bind(ctx.internalAdapter);
  const state = { fired: false, committed: false };
  vi.spyOn(adapter, method).mockImplementation(async (...args: unknown[]) => {
    const result = await original(...args);
    if (!state.fired) {
      state.fired = true;
      const change = new Promise<void>((resolve, reject) =>
        setImmediate(() =>
          db.transaction((tx) => applyVerifiedEmailChange(tx, uid, NEXT)).then(() => { state.committed = true; resolve(); }, reject)
        )
      );
      await change;
    }
    return result;
  });
  return state;
}

describe("v10-r2: Google callback vs. committed verified email change", () => {
  it("control: a session issued before the change is revoked, and the old sub is refused afterwards", async () => {
    const uid = await inviteSoldier();
    const before = await googleSignIn({ sub: "sub-a", email: OLD, email_verified: true });
    expect((await getActor(new Headers({ cookie: before.cookie })))?.id).toBe(uid);
    await db.transaction((tx) => applyVerifiedEmailChange(tx, uid, NEXT));
    const revoked = await getActor(new Headers({ cookie: before.cookie }));
    const after = await googleSignIn({ sub: "sub-a", email: OLD, email_verified: true });
    console.log("V10 CONTROL:", JSON.stringify({ revokedActor: revoked?.id ?? null, after: { signedIn: after.signedIn, error: after.error }, links: await links(uid) }));
    expect(revoked).toBeNull();
    expect(after.signedIn).toBe(false);
  });

  it("A (soldier): change commits after updateAccount; the disconnected sub still gets a session accepted by getActor", async () => {
    const uid = await inviteSoldier();
    await googleSignIn({ sub: "sub-a", email: OLD, email_verified: true });
    const st = await pauseAfter("updateAccount", uid);
    const raced = await googleSignIn({ sub: "sub-a", email: OLD, email_verified: true });
    vi.restoreAllMocks();
    const actor = raced.cookie ? await getActor(new Headers({ cookie: raced.cookie })) : null;
    const r = await row(uid);
    const sessions = await db.select({ epoch: session.securityEpoch, expiresAt: session.expiresAt }).from(session).where(eq(session.userId, uid));
    const hours = sessions.map((s) => Math.round((s.expiresAt.getTime() - Date.now()) / 3_600_000));
    console.log("V10 RACE A:", JSON.stringify({ ...st, signedIn: raced.signedIn, actorIsMember: actor?.id === uid, email: r.email, epoch: r.securityEpoch, sessionEpochs: sessions.map((s) => s.epoch), sessionHoursLeft: hours, links: await links(uid) }));
    expect(st.committed).toBe(true);
    expect(actor?.id).toBe(uid);
    expect(await links(uid)).toEqual([]);
  });

  it("A (technical account): same schedule on the technical role", async () => {
    const tech = await createInvitedAccount({ name: "v10 tech", role: "technical", email: OLD });
    await googleSignIn({ sub: "sub-t", email: OLD, email_verified: true });
    const st = await pauseAfter("updateAccount", tech.id);
    const raced = await googleSignIn({ sub: "sub-t", email: OLD, email_verified: true });
    vi.restoreAllMocks();
    const actor = raced.cookie ? await getActor(new Headers({ cookie: raced.cookie })) : null;
    console.log("V10 RACE A TECH:", JSON.stringify({ ...st, signedIn: raced.signedIn, actorRole: actor?.role ?? null, email: (await row(tech.id)).email, links: await links(tech.id) }));
    expect(actor?.role).toBe("technical");
  });

  it("B (never linked): change commits after findUserByEmail(old); a Google account for the old address gets a lasting link", async () => {
    const uid = await inviteSoldier();
    const st = await pauseAfter("findUserByEmail", uid);
    const raced = await googleSignIn({ sub: "sub-old-mailbox", email: OLD, email_verified: true });
    vi.restoreAllMocks();
    const linksAfter = await links(uid);
    const later = await googleSignIn({ sub: "sub-old-mailbox", email: OLD, email_verified: true });
    const laterActor = later.cookie ? await getActor(new Headers({ cookie: later.cookie })) : null;
    // The legitimate holder of the new address cannot link Google now.
    const legit = await googleSignIn({ sub: "sub-new-owner", email: NEXT, email_verified: true });
    console.log("V10 RACE B:", JSON.stringify({ ...st, signedIn: raced.signedIn, email: (await row(uid)).email, links: linksAfter, laterActorIsMember: laterActor?.id === uid, legitNewAddress: { signedIn: legit.signedIn, error: legit.error } }));
    expect(st.committed).toBe(true);
    expect(linksAfter).toEqual([{ accountId: "sub-old-mailbox" }]);
    expect(laterActor?.id).toBe(uid);
  });
});
