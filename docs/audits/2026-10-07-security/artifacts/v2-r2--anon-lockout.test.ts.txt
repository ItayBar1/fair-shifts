import { randomUUID } from "node:crypto";
import { beforeEach, afterAll, describe, expect, it } from "vitest";
import { eq, sql } from "drizzle-orm";
import { db, pool } from "../src/server/db";
import { user, session, loginCode } from "../src/server/auth-schema";
import { soldiers, soldierContacts, balances } from "../src/server/schema";
import {
  createInvitedAccount,
  unlockAccount,
  issueRecoveryCodes,
  type Actor,
} from "../src/server/auth/accounts";
import { soldier } from "../tests/fixtures";
import { POST } from "../src/app/api/auth/[...all]/route";

if (
  !process.env.TEST_DATABASE_URL ||
  !new URL(process.env.TEST_DATABASE_URL).pathname.endsWith("_test")
)
  throw new Error("needs *_test database");

const ORIGIN = "http://localhost:3000";
const log: string[] = [];
let technical: Actor;
let managerId: string;
const managerEmail = "manager@example.invalid";
const techEmail = "technical@example.invalid";

// Anonymous, cookie-less request: only the Origin header is set, as any
// non-browser client can do.
function anon(path: string, body: unknown) {
  return POST(
    new Request(`${ORIGIN}/api/auth/${path}`, {
      method: "POST",
      headers: { origin: ORIGIN, "content-type": "application/json" },
      body: JSON.stringify(body),
    })
  );
}
async function attack(email: string) {
  const statuses: number[] = [];
  statuses.push((await anon("request-code", { email })).status);
  for (let i = 0; i < 5; i++)
    statuses.push(
      (await anon("verify-code", { email, code: "000000x" })).status
    );
  return statuses;
}
async function row(id: string) {
  return (await db.select().from(user).where(eq(user.id, id)))[0];
}

beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`
  );
  const tech = await createInvitedAccount({
    name: "tech",
    role: "technical",
    email: techEmail,
  });
  technical = { id: tech.id, name: tech.name, role: "technical", securityEpoch: 1 };
  const sid = randomUUID();
  await db.insert(soldiers).values({
    id: sid,
    name: "mgr",
    personalNumber: "00002",
    data: soldier({ id: sid, name: "mgr", personalNumber: "00002" }),
  });
  await db.insert(soldierContacts).values({ soldierId: sid, email: managerEmail });
  await db.insert(balances).values({ soldierId: sid });
  managerId = (
    await createInvitedAccount({
      name: "mgr",
      role: "manager",
      email: managerEmail,
      soldierId: sid,
    })
  ).id;
});
afterAll(async () => {
  console.log("AUDIT-LOG\n" + log.join("\n"));
  await pool.end();
});

describe("anonymous route-level lockout", () => {
  it("locks a manager, revokes its session, and relocks right after release", async () => {
    await db.insert(session).values({
      id: "victim-session",
      userId: managerId,
      token: "synthetic-victim",
      securityEpoch: 1,
      expiresAt: new Date(Date.now() + 86_400_000),
    });
    const first = await attack(managerEmail);
    const locked = await row(managerId);
    const sessions = await db.select().from(session).where(eq(session.userId, managerId));
    log.push(
      `manager attack#1 statuses=${first.join(",")} lockedAt=${!!locked.lockedAt} failedAttempts=${locked.failedAttempts} epoch=${locked.securityEpoch} sessions=${sessions.length}`
    );
    expect(first[0]).toBe(200);
    expect(locked.lockedAt).not.toBeNull();
    expect(sessions).toHaveLength(0);

    await unlockAccount(technical, managerId);
    const codes = await db.select().from(loginCode).where(eq(loginCode.userId, managerId));
    log.push(`after technical unlock: lockedAt=${!!(await row(managerId)).lockedAt} loginCodes=${codes.length}`);

    const second = await attack(managerEmail);
    const relocked = await row(managerId);
    log.push(
      `manager attack#2 (immediately after unlock) statuses=${second.join(",")} lockedAt=${!!relocked.lockedAt}`
    );
    expect(second[0]).toBe(200);
    expect(relocked.lockedAt).not.toBeNull();
  });

  it("locks the technical account; recovery-code release is immediately relockable", async () => {
    const statuses = await attack(techEmail);
    log.push(`technical attack#1 statuses=${statuses.join(",")} lockedAt=${!!(await row(technical.id)).lockedAt}`);
    expect((await row(technical.id)).lockedAt).not.toBeNull();
    const [code] = await db.transaction((tx) => issueRecoveryCodes(technical.id, tx));
    const rec = await anon("recovery", { email: techEmail, code });
    log.push(`technical recovery status=${rec.status} lockedAt=${!!(await row(technical.id)).lockedAt}`);
    const again = await attack(techEmail);
    log.push(`technical attack#2 statuses=${again.join(",")} lockedAt=${!!(await row(technical.id)).lockedAt}`);
    expect((await row(technical.id)).lockedAt).not.toBeNull();
  });

  it("a request without the exact Origin is refused before any state change", async () => {
    const res = await POST(
      new Request(`${ORIGIN}/api/auth/verify-code`, {
        method: "POST",
        headers: { "content-type": "application/json" },
        body: JSON.stringify({ email: managerEmail, code: "1" }),
      })
    );
    log.push(`no-origin verify-code status=${res.status} failedAttempts=${(await row(managerId)).failedAttempts}`);
    expect(res.status).toBe(403);
  });
});
