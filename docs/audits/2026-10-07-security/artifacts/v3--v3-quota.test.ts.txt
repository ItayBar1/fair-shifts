import { randomUUID } from "node:crypto";
import { afterAll, beforeEach, describe, expect, it } from "vitest";
import { eq, sql } from "drizzle-orm";
import { db, pool } from "../src/server/db";
import { emailOutbox, emailQuota } from "../src/server/auth-schema";
import { balances, soldierContacts, soldiers } from "../src/server/schema";
import { createInvitedAccount } from "../src/server/auth/accounts";
import { requestCode } from "../src/server/auth/otp";
import {
  codeDeliveryDelayed,
  deliverNextEmail,
  quotaDay,
} from "../src/server/operations/email";
import { soldier } from "../tests/fixtures";

if (
  !process.env.TEST_DATABASE_URL ||
  process.env.TEST_DATABASE_URL !== process.env.DATABASE_URL ||
  !new URL(process.env.TEST_DATABASE_URL).pathname.endsWith("_test")
)
  throw new Error("requires *_test database");

const victimA = "dummy-a@example.invalid";
const victimB = "dummy-b@example.invalid";

async function invite(name: string, email: string, personalNumber: string) {
  const id = randomUUID();
  await db.insert(soldiers).values({
    id,
    name,
    personalNumber,
    data: soldier({ id, name, personalNumber }),
  });
  await db.insert(soldierContacts).values({ soldierId: id, email });
  await db.insert(balances).values({ soldierId: id });
  return createInvitedAccount({ name, role: "soldier", email, soldierId: id });
}
async function used(now: Date) {
  const [q] = await db
    .select()
    .from(emailQuota)
    .where(eq(emailQuota.day, quotaDay(now)));
  return q?.used ?? 0;
}

beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`
  );
});
afterAll(async () => pool.end());

describe("v3: request-code spends the shared daily quota", () => {
  it("one address requested every 61 s spends the whole quota, then another account's code is held", async () => {
    const accountA = await invite("דמה א", victimA, "91001");
    const accountB = await invite("דמה ב", victimB, "91002");
    const STEP = 61_000;
    const SPAN = 310 * STEP;
    // Injected times must be after the DB's default next_attempt_at (real now)
    // and inside one quota day.
    let base = new Date(Date.now() + 60_000);
    if (quotaDay(base) !== quotaDay(new Date(base.getTime() + SPAN)))
      base = new Date(`${quotaDay(new Date(base.getTime() + SPAN))}T00:01:00Z`);
    const day = quotaDay(base);

    const sentTo: string[] = [];
    const transport = async (m: { to: string; eventKey: string }) => {
      sentTo.push(m.to);
      return `synthetic-${m.eventKey}`;
    };

    // Control: the per-account 60 s resend window does reject a faster repeat.
    await requestCode(victimA, base);
    await expect(
      requestCode(victimA, new Date(base.getTime() + 30_000))
    ).rejects.toMatchObject({ status: 429 });
    expect(await deliverNextEmail(transport, base)).toEqual({ status: "sent" });
    expect(await used(base)).toBe(1);

    let i = 1;
    let last = base;
    for (; i < 400; i++) {
      const t = new Date(base.getTime() + i * STEP);
      last = t;
      const before = await used(t);
      if (before >= 300) break;
      const answer = await requestCode(victimA, t);
      expect(answer).toEqual({ success: true });
      const r = await deliverNextEmail(transport, t);
      expect(r).toEqual({ status: "sent" });
      expect(await used(t)).toBe(before + 1);
    }
    expect(quotaDay(last)).toBe(day);
    expect(await used(last)).toBe(300);
    expect(sentTo.length).toBe(300);
    expect(new Set(sentTo)).toEqual(new Set([victimA]));
    const requests = i; // codes requested for A (including the first)
    const elapsedMin = (last.getTime() - base.getTime()) / 60_000;

    // Another, uninvolved account now asks for a code.
    const t = new Date(last.getTime() + 1000);
    expect(await codeDeliveryDelayed(db, t)).toBe(true);
    expect(await requestCode(victimB, t)).toEqual({
      success: true,
      mailDelayed: true,
    });
    expect(await deliverNextEmail(transport, t)).toEqual({ status: "idle" });
    const [heldB] = await db
      .select()
      .from(emailOutbox)
      .where(eq(emailOutbox.recipientAccountId, accountB.id));
    expect(heldB).toMatchObject({
      kind: "login-code",
      status: "pending",
      error: "quota_waiting",
    });
    expect(sentTo.length).toBe(300);
    console.log(
      JSON.stringify({
        quotaDay: day,
        requestsForA: requests,
        deliveredToA: sentTo.length,
        quotaUsed: await used(t),
        elapsedMinutes: Math.round(elapsedMin),
        codeDeliveryDelayed: true,
        accountBCode: { status: heldB.status, error: heldB.error },
        accountAId: typeof accountA.id,
      })
    );
  }, 300_000);
});
