import { randomUUID } from "node:crypto";
import { beforeEach, afterAll, describe, expect, it } from "vitest";
import { and, eq, sql } from "drizzle-orm";
import { db, pool } from "../src/server/db";
import { user, session, account, emailOutbox } from "../src/server/auth-schema";
import { soldiers, soldierContacts, balances } from "../src/server/schema";
import { createInvitedAccount, type Actor } from "../src/server/auth/accounts";
import { requestCode, verifyCode } from "../src/server/auth/otp";
import { openSecret } from "../src/server/operations/email";
import { executeAction } from "../src/server/actions";
import { soldier } from "../tests/fixtures";

if (!process.env.TEST_DATABASE_URL?.endsWith("_test"))
  throw new Error("needs *_test db");

let technical: Actor;
let managerA: Actor;
let managerB: Actor;
const B_OLD = "manager-b@example.invalid";
const ATTACKER = "attacker-of-a@example.invalid";

async function invite(name: string, role: "soldier" | "manager", email: string, pn: string) {
  const id = randomUUID();
  await db.insert(soldiers).values({ id, name, personalNumber: pn, data: soldier({ id, name, personalNumber: pn }) });
  await db.insert(soldierContacts).values({ soldierId: id, email });
  await db.insert(balances).values({ soldierId: id });
  const u = await createInvitedAccount({ name, role, email, soldierId: id });
  return { id: u.id, name: u.name, role, soldierId: id, securityEpoch: 1 } as Actor;
}
function cmd(actor: Actor, type: string, payload: Record<string, unknown>, expectedVersion?: number) {
  return executeAction(actor, { type, payload, expectedVersion, idempotencyKey: randomUUID() });
}

beforeEach(async () => {
  await db.execute(sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`);
  const tech = await createInvitedAccount({ name: "tech", role: "technical", email: "technical@example.invalid" });
  technical = { id: tech.id, name: tech.name, role: "technical", securityEpoch: 1 };
  managerA = await invite("Manager A", "manager", "manager-a@example.invalid", "00011");
  managerB = await invite("Manager B", "manager", B_OLD, "00012");
});
afterAll(async () => pool.end());

describe("v5: peer manager email change", () => {
  it("manager A moves manager B's login email and signs in as B", async () => {
    await db.insert(account).values({ id: "b-google", userId: managerB.id, providerId: "google", accountId: "b-sub" });
    await db.insert(session).values({ id: "b-session", userId: managerB.id, token: "b-token", securityEpoch: 1, expiresAt: new Date(Date.now() + 3_600_000) });

    // Controls: sibling manager-on-manager operations are refused.
    await expect(cmd(managerA, "account.unlock", { id: managerB.id }, 1)).rejects.toThrow();
    await expect(cmd(technical, "account.email.request", { soldierId: managerB.soldierId, email: "x@example.invalid", reason: "synthetic" }, 1)).rejects.toThrow();
    console.log("CONTROL unlock(B) by A rejected; technical account.email.request rejected");

    await cmd(managerA, "account.email.request", { soldierId: managerB.soldierId, email: ATTACKER, reason: "synthetic" }, 1);
    const [mail] = await db.select().from(emailOutbox).where(and(eq(emailOutbox.kind, "email-change"), eq(emailOutbox.recipientAccountId, managerB.id)));
    console.log("EMAIL-CHANGE MAIL destination:", mail.destination);
    expect(mail.destination).toBe(ATTACKER);
    const code = openSecret(mail.encryptedSecret!);
    await cmd(managerA, "account.email.confirm", { soldierId: managerB.soldierId, code, disconnectGoogle: true }, 1);

    const [b] = await db.select().from(user).where(eq(user.id, managerB.id));
    console.log("B AFTER:", JSON.stringify({ email: b.email, role: b.role, emailVerified: b.emailVerified, securityEpoch: b.securityEpoch }));
    expect(b).toMatchObject({ email: ATTACKER, role: "manager", securityEpoch: 2 });
    const bSessions = await db.select().from(session).where(eq(session.userId, managerB.id));
    const bLinks = await db.select().from(account).where(eq(account.userId, managerB.id));
    console.log("B sessions:", bSessions.length, "B google links:", bLinks.length);
    expect(bSessions).toHaveLength(0);
    expect(bLinks).toHaveLength(0);

    // Old address no longer receives a login code; attacker address does.
    await requestCode(B_OLD);
    await requestCode(ATTACKER);
    const loginMails = (await db.select().from(emailOutbox)).filter((r) => r.kind === "login-code");
    console.log("login-code mails:", loginMails.length, "destinations:", JSON.stringify(loginMails.map((r) => r.destination)), "recipient is B:", loginMails.every((r) => r.recipientAccountId === managerB.id));
    expect(loginMails).toHaveLength(1);
    const outcome = await verifyCode(ATTACKER, openSecret(loginMails[0].encryptedSecret!));
    console.log("verifyCode(attacker address) ->", JSON.stringify({ userId: outcome.user.id, isB: outcome.user.id === managerB.id, role: outcome.user.role, epoch: outcome.epoch }));
    expect(outcome.user.id).toBe(managerB.id);
    expect(outcome.user.role).toBe("manager");
  });
});
