// h-mail-r2: bounded check of login-code sealing and outbox secret lifetime (synthetic data only).
import { randomUUID } from "node:crypto";
import { afterAll, beforeEach, describe, expect, it } from "vitest";
import { eq, sql } from "drizzle-orm";
import { db, pool } from "../src/server/db";
import { emailOutbox, loginCode } from "../src/server/auth-schema";
import { balances, soldierContacts, soldiers } from "../src/server/schema";
import {
  createInvitedAccount,
  revokeAccess,
} from "../src/server/auth/accounts";
import { requestCode, verifyCode } from "../src/server/auth/otp";
import {
  deliverNextEmail,
  openSecret,
  sealSecret,
} from "../src/server/operations/email";
import { soldier } from "../tests/fixtures";

if (
  !process.env.TEST_DATABASE_URL ||
  !new URL(process.env.TEST_DATABASE_URL).pathname.endsWith("_test")
)
  throw new Error("dedicated *_test database required");

const email = "seal-member@example.invalid";
let accountId = "";
beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`
  );
  const id = randomUUID();
  await db.insert(soldiers).values({
    id,
    name: "seal",
    personalNumber: "39001",
    data: soldier({ id, name: "seal", personalNumber: "39001" }),
  });
  await db.insert(soldierContacts).values({ soldierId: id, email });
  await db.insert(balances).values({ soldierId: id });
  const row = await createInvitedAccount({
    name: "seal",
    role: "soldier",
    email,
    soldierId: id,
  });
  accountId = row.id;
});
afterAll(async () => pool.end());

describe("h-mail-r2 sealing", () => {
  it("seal/open: random IV, GCM integrity, truncated-tag behaviour", () => {
    const a = sealSecret("123456");
    const b = sealSecret("123456");
    expect(a).not.toBe(b);
    expect(a.split(".")[0]).not.toBe(b.split(".")[0]);
    expect(openSecret(a)).toBe("123456");
    const [iv, tag, ct] = a.split(".");
    const flipped = Buffer.from(ct, "base64url");
    flipped[0] ^= 1;
    expect(() =>
      openSecret([iv, tag, flipped.toString("base64url")].join("."))
    ).toThrow();
    const short = Buffer.from(tag, "base64url").subarray(0, 4);
    let truncated: string;
    try {
      truncated = openSecret([iv, short.toString("base64url"), ct].join("."));
    } catch (error) {
      truncated = `throws:${(error as Error).message}`;
    }
    console.log(`H-MAIL-R2 truncated 4-byte tag result: ${truncated}`);
    console.log(`H-MAIL-R2 node version: ${process.version}`);
  });

  it("login code is never stored in clear and its sealed copy is cleared on revoke/skip", async () => {
    await requestCode(email);
    const [message] = await db
      .select()
      .from(emailOutbox)
      .where(eq(emailOutbox.recipientAccountId, accountId));
    expect(message.kind).toBe("login-code");
    expect(message.body).toContain("{{CODE}}");
    expect(message.destination).toBeNull();
    const code = openSecret(message.encryptedSecret!);
    expect(code).toMatch(/^\d{6}$/);
    // Search every table row for the clear code.
    const tables = await db.execute(
      sql`select tablename from pg_tables where schemaname = 'public'`
    );
    const hits: string[] = [];
    for (const { tablename } of tables.rows as { tablename: string }[]) {
      const rows = await db.execute(
        sql.raw(
          `select row_to_json(t)::text as j from "${tablename}" t where row_to_json(t)::text like '%${code}%'`
        )
      );
      for (const r of rows.rows as { j: string }[])
        hits.push(`${tablename}: ${r.j.slice(0, 200)}`);
    }
    console.log(`H-MAIL-R2 clear-code hits: ${JSON.stringify(hits)}`);
    expect(hits).toEqual([]);

    // Verified code: a still-pending message is skipped at delivery and its secret cleared.
    await verifyCode(email, code);
    const sent: string[] = [];
    const result = await deliverNextEmail(async (m) => {
      sent.push(m.text);
      return "synthetic";
    });
    const [after] = await db
      .select()
      .from(emailOutbox)
      .where(eq(emailOutbox.id, message.id));
    console.log(
      `H-MAIL-R2 after verify+deliver: result=${result.status} status=${after.status} error=${after.error} secretCleared=${after.encryptedSecret === null} sent=${sent.length}`
    );
    expect(sent).toEqual([]);
    expect(after.encryptedSecret).toBeNull();
  });

  it("revokeAccess cancels a pending login-code message and drops the sealed code", async () => {
    await requestCode(email);
    await db.transaction((tx) => revokeAccess(tx, accountId));
    const [message] = await db
      .select()
      .from(emailOutbox)
      .where(eq(emailOutbox.recipientAccountId, accountId));
    const [code] = await db
      .select()
      .from(loginCode)
      .where(eq(loginCode.userId, accountId));
    console.log(
      `H-MAIL-R2 after revoke: status=${message.status} secretCleared=${message.encryptedSecret === null} bodyEmpty=${message.body === ""} loginCodeRow=${code ? "present" : "deleted"}`
    );
    expect(message.status).toBe("cancelled");
    expect(message.encryptedSecret).toBeNull();
  });
});
