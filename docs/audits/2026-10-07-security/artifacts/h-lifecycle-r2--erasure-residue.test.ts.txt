// Audit harness (h-lifecycle-r2): synthetic people only.
const DAY = 86_400_000;
let manager: Actor;
let member: Actor;
let other: Actor;

async function invite(
  name: string,
  role: "soldier" | "manager",
  personalNumber: string
): Promise<Actor> {
  const soldierId = randomUUID();
  await db.insert(soldiers).values({
    id: soldierId,
    name,
    personalNumber,
    data: soldier({ id: soldierId, name, personalNumber }),
  });
  await db.insert(soldierContacts).values({
    soldierId,
    email: `${personalNumber}@example.invalid`,
    phone: `050${personalNumber}`,
    address: `addr ${personalNumber}`,
  });
  await db.insert(balances).values({ soldierId });
  const row = await createInvitedAccount({
    name,
    role,
    email: `${personalNumber}@example.invalid`,
    soldierId,
  });
  return { id: row.id, name, role, soldierId, securityEpoch: 1 };
}
async function command(
  actor: Actor,
  type: string,
  payload: Record<string, unknown>,
  expectedVersion?: number,
  idempotencyKey: string = randomUUID()
) {
  return (await executeAction(actor, {
    type,
    payload,
    expectedVersion,
    idempotencyKey,
  })) as { id: string; version: number } & Record<string, unknown>;
}
async function soldierRowById(id: string) {
  const [row] = await db.select().from(soldiers).where(eq(soldiers.id, id));
  return row;
}
async function dutyRow(id: string) {
  const [row] = await db.select().from(duties).where(eq(duties.id, id));
  return row;
}
async function publishedDuty(seats: Actor[], startInDays = 2) {
  const type = await command(manager, "dutyType.save", {
    name: `type ${randomUUID().slice(0, 6)}`,
    pricing: { mode: "fixed", base: 4 },
    roles: [{ name: "תורן", count: seats.length }],
  });
  const duty = await command(manager, "duty.create", {
    typeId: type.id,
    name: "duty",
    start: new Date(Date.now() + startInDays * DAY).toISOString(),
    end: new Date(Date.now() + (startInDays + 1) * DAY).toISOString(),
  });
  const created = await dutyRow(duty.id);
  let version = 1;
  for (const [index, person] of seats.entries())
    await command(
      manager,
      "duty.assign",
      {
        dutyId: duty.id,
        slotId: created.data.slots[index].id,
        soldierId: person.soldierId,
      },
      version++
    );
  await command(manager, "duty.publish", { id: duty.id, confirmed: true }, version);
  return dutyRow(duty.id);
}
async function remove(actor: Actor, soldierId: string) {
  const row = await soldierRowById(soldierId);
  const preview = (await command(
    actor,
    "soldier.delete.preview",
    { id: soldierId },
    row.version
  )) as unknown as { previewToken: string };
  const now = await soldierRowById(soldierId);
  return command(
    actor,
    "soldier.delete",
    {
      id: soldierId,
      previewToken: preview.previewToken,
      reason: "released",
      confirmed: true,
    },
    now.version
  );
}

beforeEach(async () => {
  await db.execute(
    sql`truncate table auth_user, soldiers, duty_types, unit_lock, email_quota, operations_state, command_results cascade`
  );
  manager = await invite("manager one", "manager", "00201");
  member = await invite("member to delete", "soldier", "00203");
  other = await invite("other soldier", "soldier", "00204");
});
afterAll(async () => pool.end());

describe("audit: erasure of a transfer decision reason", () => {
  it("scrubs the request and site notices but leaves the reason in the counterpart's email_outbox row", async () => {
    const REASON = "AUDIT-SECRET-REJECT-REASON-7731";
    const duty = await publishedDuty([other], 2);
    const [seat] = await db
      .select()
      .from(assignments)
      .where(eq(assignments.soldierId, other.soldierId!));
    const offer = await command(
      other,
      "transfer.offer",
      { assignmentId: seat.id, candidateIds: [member.soldierId] },
      seat.version
    );
    // The duty starts, so the acceptance waits for a manager.
    const row = await dutyRow(duty.id);
    await db
      .update(duties)
      .set({
        data: {
          ...row.data,
          start: new Date(Date.now() - 3_600_000).toISOString(),
          end: new Date(Date.now() + DAY).toISOString(),
        },
      })
      .where(eq(duties.id, duty.id));
    const accepted = await command(
      member,
      "transfer.respond",
      { id: offer.id, decision: "accept", confirmed: true },
      offer.version
    );
    expect(accepted.status).toBe("awaiting_manager");
    await command(
      manager,
      "transfer.decide",
      { id: offer.id, decision: "reject", reason: REASON },
      accepted.version
    );
    const before = await db
      .select()
      .from(emailOutbox)
      .where(sql`position(${REASON} in ${emailOutbox.body}) > 0`);
    console.log(
      "AUDIT outbox rows quoting reason before deletion:",
      before.map((m) => ({
        recipientIsMember: m.recipientAccountId === member.id,
        recipientIsOther: m.recipientAccountId === other.id,
        status: m.status,
        kind: m.kind,
      }))
    );

    await remove(manager, member.soldierId!);

    const recordHits = await db
      .select()
      .from(records)
      .where(sql`position(${REASON} in ${records.data}::text) > 0`);
    const resultHits = await db
      .select()
      .from(commandResults)
      .where(sql`position(${REASON} in ${commandResults.result}::text) > 0`);
    const after = await db
      .select()
      .from(emailOutbox)
      .where(sql`position(${REASON} in ${emailOutbox.body}) > 0`);
    console.log(
      "AUDIT after deletion: records quoting reason =",
      recordHits.map((r) => r.kind),
      "command_results quoting reason =",
      resultHits.length,
      "outbox rows quoting reason =",
      after.map((m) => ({
        recipientIsMember: m.recipientAccountId === member.id,
        recipientIsOther: m.recipientAccountId === other.id,
        status: m.status,
        kind: m.kind,
      }))
    );
    // Primary and site copies are scrubbed as the deletion promises.
    expect(recordHits).toHaveLength(0);
    // The mail copy queued for the counterpart still quotes the reason.
    expect(after.filter((m) => m.recipientAccountId === other.id).length).toBe(1);
  });
});

describe("audit: erasure of contact values in a stored import preview", () => {
  it("leaves an earlier phone and address in the manager's stored import.preview result", async () => {
    const OLD_PHONE = "0507770001";
    const OLD_ADDRESS = "AUDIT-OLD-ADDRESS-4411";
    const previewKey = randomUUID();
    const preview = await command(
      manager,
      "import.preview",
      {
        filename: "intake.xlsx",
        rows: [
          {
            rowNumber: 2,
            values: {
              personalNumber: "00999",
              name: "imported soldier",
              email: "imported@example.invalid",
              phone: OLD_PHONE,
              address: OLD_ADDRESS,
            },
          },
        ],
      },
      undefined,
      previewKey
    );
    const applied = await command(
      manager,
      "import.apply",
      { id: preview.id, confirmed: true, overwriteConfirmed: true, reason: "intake" },
      preview.version
    );
    const [person] = await db
      .select()
      .from(soldiers)
      .where(eq(soldiers.personalNumber, "00999"));
    expect(person).toBeDefined();
    void applied;
    // Contact details change later in the soldier's service.
    await command(
      manager,
      "soldier.update",
      {
        id: person.id,
        name: person.name,
        personalNumber: person.personalNumber,
        phone: "0507770002",
        address: "new address",
      },
      person.version
    );
    // The login (and contact) email is changed through the verified flow.
    const fresh = await soldierRowById(person.id);
    await command(
      manager,
      "account.email.request",
      { soldierId: person.id, email: "changed@example.invalid", reason: "new mailbox" },
      fresh.version
    );
    const [mail] = await db
      .select()
      .from(emailOutbox)
      .where(eq(emailOutbox.kind, "email-change"));
    const code = openSecret(mail.encryptedSecret!);
    await command(
      manager,
      "account.email.confirm",
      { soldierId: person.id, code, disconnectGoogle: true },
      (await soldierRowById(person.id)).version
    );
    const [contactNow] = await db
      .select()
      .from(soldierContacts)
      .where(eq(soldierContacts.soldierId, person.id));
    console.log("AUDIT contact email after verified change:", contactNow.email);
    await remove(manager, person.id);
    const [stored] = await db
      .select()
      .from(commandResults)
      .where(eq(commandResults.requestKey, previewKey));
    const text = JSON.stringify(stored.result);
    console.log(
      "AUDIT stored import.preview result after deletion contains old phone:",
      text.includes(OLD_PHONE),
      "old address:",
      text.includes(OLD_ADDRESS),
      "old email:",
      text.includes("imported@example.invalid"),
      "personal number:",
      text.includes("00999"),
      "erasedAt marker:",
      "erasedAt" in (stored.result as object)
    );
    expect(text).toContain(OLD_PHONE);
    expect(text).toContain(OLD_ADDRESS);
    // A replay of the same request returns the stored result to the manager.
    const replay = await executeAction(manager, {
      type: "import.preview",
      payload: {
        filename: "intake.xlsx",
        rows: [
          {
            rowNumber: 2,
            values: {
              personalNumber: "00999",
              name: "imported soldier",
              email: "imported@example.invalid",
              phone: OLD_PHONE,
              address: OLD_ADDRESS,
            },
          },
        ],
      },
      idempotencyKey: previewKey,
    });
    console.log(
      "AUDIT replayed result contains old address:",
      JSON.stringify(replay).includes(OLD_ADDRESS)
    );
  });
});
